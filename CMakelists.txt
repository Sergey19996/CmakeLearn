cmake_minimum_required(VERSION 3.13)

set(CMAKE_TOOLCHAIN_FILE C:/vcpkg/scripts/buildsystems/vcpkg.cmake)

project(OLAS VERSION 1.3)

set(CMAKE_INSTALL_PREFIX ${PROJECT_BINARY_DIR})



#adder use logic
option(USE_ADDER "A simple library for adding two floats" ON)

if(USE_ADDER)
    add_subdirectory(Adder)

    # Добавляем Adder в список библиотек и инклюдов
    list(APPEND EXTRA_LIBS adder)
    list(APPEND EXTRA_INCLUDES "${PROJECT_SOURCE_DIR}/Adder")
endif()



find_package(glad CONFIG REQUIRED)
if(glad_FOUND)
message(STATUS " glut  found ${glad_LIBRARIES}")
else()
message(FATAL_ERROR "glad not found! Укажите путь через -DCMAKE_PREFIX_PATH=...")
endif()



message(STATUS "Компилятор C++: ${CMAKE_CXX_COMPILER}")
message(STATUS "Компилятор C: ${CMAKE_C_COMPILER}")

configure_file(OLASConfig.h.in OLASConfig.h)


add_executable(${PROJECT_NAME} main.cpp)


target_include_directories(${PROJECT_NAME} PUBLIC ${PROJECT_BINARY_DIR})




# ==============================
#       DOWNLOAD ALL THE SUBMODULES
# ==============================
find_package(Git QUIET)

if(GIT_FOUND AND EXISTS "${PROJECT_SOURCE_DIR}/.git")
    # Опция: обновлять подмодули во время сборки
    option(GIT_SUBMODULE "Check submodules during build" ON)

    if(GIT_SUBMODULE)
        message(STATUS "Submodule update")
        execute_process(
            COMMAND ${GIT_EXECUTABLE} submodule update --init --recursive
            WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            RESULT_VARIABLE GIT_SUBMOD_RESULT
        )

        if(NOT GIT_SUBMOD_RESULT EQUAL "0")
            message(FATAL_ERROR "git submodule update --init failed with ${GIT_SUBMOD_RESULT}, please checkout submodules")
        endif()
    endif()
endif()

# ==============================
#       CHECK ALL THE SUBMODULES
# ==============================
if(NOT EXISTS "${PROJECT_SOURCE_DIR}/external/glfw/CMakeLists.txt")
    message(FATAL_ERROR "The glfw submodules was not downloaded! GIT SUBMODULE was turned off or failed. Please update submodules and try again.")
endif()

add_subdirectory(external/glfw)

target_include_directories(${PROJECT_NAME}
 PRIVATE ${EXTRA_INCLUDES})



target_link_libraries(${PROJECT_NAME}  glfw ${EXTRA_LIBS}  glad::glad)

install(TARGETS ${PROJECT_NAME} DESTINATION bin)
install(FILES "${PROJECT_BINARY_DIR}/OLASConfig.h" DESTINATION include)

include(InstallRequiredSystemLibraries)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/Licence.txt")
set(CPACK_PACKAGE_VERSION_MAJOR "${TUTORIAL_VERSION_MAJOR}")
set(CPACK_PACKAGE_VERSION_MINOR "${TUTORIAL_VERSION_MINOR}")
include(CPack)
